# 最小部署测试代码
```python
# ====================== 顶部：屏蔽所有无关警告 ======================
import os

os.environ["HF_HUB_DISABLE_SYMLINKS_WARNING"] = "1"
os.environ["TRANSFORMERS_NO_ADVISORY_WARNINGS"] = "1"
import warnings

warnings.filterwarnings("ignore")
# =================================================================

import os
import torch
from transformers import AutoTokenizer

# ===== STEP 1: torch =====
print("===== STEP 1: torch =====")
print(f"torch version: {torch.__version__}")
print(f"cuda available: {torch.cuda.is_available()}")
device = torch.device("cuda")

# ===== STEP 2: 模型导入 =====
print("===== STEP 2: 模型导入 =====")
from lerobot.policies.smolvla.modeling_smolvla import SmolVLAPolicy

# modeling_smolvla是模型核心实现文件（.py），SmolVLAPolicy：SmolVLA 模型的对外接口类（核心！），封装了模型所有功能。
print("SmolVLAPolicy import OK")

# ===== STEP 3: 模型路径 =====
model_dir = os.path.abspath("./pretrained_model")
print(f"model_dir: {model_dir}")

# ===== STEP 4: 官方一键加载模型 =====
print("===== STEP 4: 一键加载模型 =====")
policy = SmolVLAPolicy.from_pretrained(model_dir)
policy.to(device)
# 将策略模型的所有参数和缓冲区移动到之前定义的 device（GPU）上。
policy.eval()
# 将模型设置为评估模式。这会关闭 Dropout 等只在训练时启用的层，确保推理时行为正确。
print("模型加载成功 ✅")

# ===== STEP 5: 分词器 =====
print("===== STEP 5: 分词器 =====")
tokenizer = AutoTokenizer.from_pretrained("HuggingFaceTB/SmolVLM2-500M-Video-Instruct")
print("Tokenizer loaded")

# ===== STEP 6: 构造完整标准输入 =====
print("===== STEP 6: 构造输入 =====")
text = "Pick up the red cube and place it in the blue box"
# 同时获取 tokens 和 attention_mask
token_output = tokenizer(
    text,
    return_tensors="pt",
    # 返回 PyTorch 张量格式。
    padding=True,
    # 对不足 max_length 的序列进行填充
    truncation=True,
    # 如果文本长度超过 max_length 则截断。
    max_length=77
    # 设置最大 token 长度为 77（符合模型上下文长度限制）。
    # 分词结果 token_output 是一个字典，包含 input_ids
    # 和 attention_mask 等字段，
    # 然后立即通过 .to(device) 将这些张量移动到 GPU。
).to(device)

#  终极修复：将 attention_mask 转为 布尔类型(bool)
dummy_obs = {
    # 创建一个字典 dummy_obs，代表一次观察输入，用于模拟机器人环境中的观测数据。
    "observation.images.top": torch.randn(1, 3, 512, 512, device=device),
    # 键为 "observation.images.top"，值是形状 (1, 3, 512, 512) 的随机张量，
    # 模拟顶视图摄像头图像（batch 大小1，3个颜色通道，分辨率512×512），并放在 GPU 上。
    "observation.images.wrist": torch.randn(1, 3, 512, 512, device=device),
    # 类似，模拟手腕摄像头图像，张量形状和属性同上。
    "observation.state": torch.randn(1, 6, device=device),
    # 键为 "observation.state"，值是形状 (1, 6) 的随机张量，
    # 模拟机器人当前状态（如关节角度，这里假设6个自由度），放在 GPU 上。
    "observation.language.tokens": token_output["input_ids"],
    # 将之前分词得到的 input_ids 放入观察字典，对应语言指令的 token 序列。
    "observation.language.attention_mask": token_output["attention_mask"].bool()
    # 将 attention_mask 转换为布尔类型后作为 "observation.language.attention_mask" 的值。
}

# ===== STEP 7: 模型推理 =====
print("===== STEP 7: 推理 =====")
with torch.no_grad():
    # 进入 torch.no_grad() 上下文，关闭梯度计算，减少内存占用并加速推理。
    action = policy.select_action(dummy_obs)
    # 调用策略对象的 select_action 方法，传入构造的观察字典 dummy_obs，
    # 模型根据输入预测出动作序列（或单个动作），返回 action 张量。

# 🎉 最终成功输出
print("🎉 Inference success! 推理完成！")
print(f"action shape: {action.shape}")
print("✅ DONE：SmolVLA 模型全流程测试完美成功！")

```
